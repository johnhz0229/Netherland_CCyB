\qtag{Summary}
\begin{frame}{The Netherlands raised its countercyclical capital buffer (CCyB) to 2\% when the rulebook said 0\% — refilling a buffer, not fighting a boom}
  \small
  \textbf{Decision:} CCyB 0\% → 1\% (ann.\ May 2022) → \hl{2\%} (ann.\ May 2023, binding 31 May 2024), together with lower buffers for systemically important banks (O-SII).\\[3pt]
  \textbf{Puzzle:} the Basel reference indicator, the credit-to-GDP gap (credit above its long-run trend), stood at \nv{−33pp} and \nv{−47pp} at the two decisions → the buffer guide implied \nv{0\%}.
  \vspace{4pt}
  \begin{columns}[T,onlytextwidth]
    \column{0.32\textwidth}
    \begin{block}{Q1 · Environment}\scriptsize
      Recovered economy, then war, energy shock and \textbf{11.6\%} inflation; ECB hikes from Jul-2022.\\[2pt]
      Credit cold; housing hot (+19\% y/y), then falling.\\[2pt]
      Banks strong (core capital ratio, CET1, 17.7\%) and more profitable as rates rose.
    \end{block}
    \column{0.32\textwidth}
    \begin{block}{Q2 · Criteria \& risks}\scriptsize
      DNB framework (2022): \textbf{2\% = “standard risk environment”}, calibrated on past crisis losses (€12bn), not on credit growth.\\[2pt]
      Insures against shocks no indicator predicts; housing left to other tools.\\[2pt]
      Keeps the March-2020 promise to restore releasable capital.
    \end{block}
    \column{0.32\textwidth}
    \begin{block}{Q3 · Implications}\scriptsize
      \textbf{Cheap:} ≤ about one fifth of big banks' CET1 headroom.\\[2pt]
      \textbf{A swap:} structural buffers cut in 2020 and 2024.\\[2pt]
      \textbf{No sign of contraction} in credit or its price.\\[2pt]
      Complements ECB tightening; reciprocated abroad; \textbf{€6.7bn} releasable.
    \end{block}
  \end{columns}
  \vspace{4pt}
  \textbf{Verdict:} \hl{normalisation, not tightening.} \quad \textbf{Trade-off:} relies on discretion and credibility; release never tested.
  \source{DNB (2020, 2022, 2023, 2026); DNB Financial Stability Reports 2022–23; BIS credit-to-GDP gaps; ECB; bank disclosures; own calculations.}
  \slidenote{1}
\end{frame}

\qtag{Prologue}
\begin{frame}{The CCyB is the only layer of bank capital designed to be switched off in a crisis}
  \begin{columns}[c,onlytextwidth]
    \column{0.44\textwidth}
    \centering
    \begin{tikzpicture}[font=\scriptsize,box/.style={draw=white,thick,minimum width=4.1cm,align=center,inner sep=2pt}]
      \node[box,fill=navy!80,text=white,minimum height=1.25cm] (p1) {Pillar 1 minimum · 4.5\%};
      \node[box,fill=navy!60,text=white,minimum height=0.65cm,above=0pt of p1] (p2r) {Pillar 2 requirement (bank-specific)};
      \node[box,fill=grey!55,minimum height=0.85cm,above=0pt of p2r] (ccob) {Capital conservation buffer · 2.5\%};
      \node[box,fill=grey!35,minimum height=0.65cm,above=0pt of ccob] (osii) {O-SII / systemic risk buffer};
      \node[box,fill=nlred,text=white,minimum height=0.65cm,above=0pt of osii,font=\scriptsize\bfseries] (ccyb) {CCyB · 0–2.5\% · \emph{releasable}};
      \node[box,draw=grey,dashed,fill=white,minimum height=0.5cm,above=0pt of ccyb] (p2g) {Pillar 2 guidance (non-binding)};
      \draw[decorate,decoration={brace,amplitude=4pt},thick] ([xshift=3pt]ccyb.north east) -- ([xshift=3pt]ccob.south east)
        node[midway,right=5pt,align=left,text width=2.3cm]{Combined buffer requirement: falling below its top → automatic limits on dividends \& bonuses (MDA)};
      \draw[nlred,very thick,dashed] ([xshift=-6pt]ccyb.north west) -- ([xshift=6pt]ccyb.north east);
      \node[left=2pt of ccyb.north west,anchor=east,text=nlred,font=\scriptsize\bfseries,align=right] {MDA\\trigger};
      \node[below=3pt of p1,font=\tiny,text=grey,text width=4.4cm,align=left] {O-SII: other systemically important institution · MDA: maximum distributable amount · \% of risk-weighted assets (RWA: assets weighted by riskiness)};
    \end{tikzpicture}
    \column{0.53\textwidth}\small
    \textbf{Design}
    \begin{itemize}\setlength\itemsep{1pt}
      \item 0–2.5\% of risk-weighted assets, on \emph{domestic} exposures
      \item Increases bind after 12 months; releases are immediate
      \item EU reciprocity: foreign banks lending in NL hold it too
    \end{itemize}
    \textbf{Why “releasable” matters}
    \begin{itemize}\setlength\itemsep{1pt}
      \item In COVID, banks close to the MDA trigger \hl{cut lending} even when \emph{allowed} to use buffers (stigma, payout limits)
      \item Only an explicitly \textbf{released} buffer is truly usable
    \end{itemize}
  \end{columns}
  \source{CRD (Arts.\ 130–136); BCBS (2024) d585; Couaillier, Lo Duca, Reghezza \& Rodriguez d'Acri (2022), ECB WP 2644. Stack not to scale.}
  \slidenote{3}
\end{frame}

\qtag{Q2 · Criteria \& risks}
\begin{frame}{DNB justified the steps by recovery, uncertainty and its 2020 promise —\\ not by credit growth}
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{block}{May 2022 · 0\% → 1\%}\small
      \itshape “When the systemic risk buffers \ldots{} were lowered in March 2020, we also announced our intention to \textbf{restore the buffers} by raising the CCyB.”\\[3pt]

\qtag{Appendix}
\begin{frame}{Glossary}
  \fontsize{7.6}{9.2}\selectfont
  \begin{columns}[T,onlytextwidth]
    \column{0.49\textwidth}
    \begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash\bfseries}p{2.3cm}X@{}}
      \toprule
      CCyB & Countercyclical capital buffer: 0–2.5\% of risk-weighted assets on domestic exposures; raised with a 12-month lag, released immediately. \\
      Positive neutral CCyB & A CCyB kept above zero when risks are neither high nor low, so it can be released for any shock. \\
      CET1 & Common Equity Tier 1: the highest-quality capital (shares, retained earnings), as \% of RWA. \\
      RWA & Risk-weighted assets: assets weighted by their riskiness (e.g.\ mortgages count less than corporate loans). \\
      Pillar 1 / P2R / P2G & Legal minimum (4.5\% CET1) / bank-specific add-on set by the supervisor / non-binding supervisory guidance. \\
      CCoB & Capital conservation buffer, 2.5\% for all banks. \\
      O-SII / SRB & Buffer for other systemically important institutions / systemic risk buffer (structural, not designed for release). \\
      MDA & Maximum distributable amount: if CET1 falls below the combined buffer requirement, payouts are automatically restricted. \\
      \bottomrule
    \end{tabularx}
    \column{0.49\textwidth}
    \begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash\bfseries}p{2.3cm}X@{}}
      \toprule
      Credit-to-GDP gap & Credit/GDP ratio minus its long-run trend; the Basel reference for the CCyB (gap 2–10pp → buffer 0–2.5\%). \\
      HP filter & Hodrick-Prescott filter: estimates a smooth trend; one-sided uses only past data (real time), two-sided also future data. \\
      Headroom & CET1 capital above a bank's requirement / MDA trigger. \\
      Reciprocity & Foreign banks must apply the Dutch CCyB rate to their Dutch exposures. \\
      Borrower-based measures & Limits on loan-to-value (LTV) and loan-to-income (LTI) of new mortgages. \\
      Risk-weight floor & Minimum average risk weight on Dutch mortgages of banks using internal models (Art.\ 458 CRR), Jan 2022 – Nov 2026. \\
      Event study / placebo & Compare NL with countries that did not change their CCyB around each date / repeat with each control as a fake “treated” country. \\
      pp / bp & Percentage point / basis point (0.01pp). \\
      \bottomrule
    \end{tabularx}
  \end{columns}
